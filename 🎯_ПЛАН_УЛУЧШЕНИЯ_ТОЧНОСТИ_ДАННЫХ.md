```python
# Что добавить в app/services/data_processing.py

def validate_data_quality(df):
    """Проверка качества данных"""
    issues = []
    
    # 1. Проверка дубликатов
    duplicates = df.duplicated().sum()
    if duplicates > 0:
        issues.append(f"⚠️ Найдено {duplicates} дубликатов")
    
    # 2. Проверка пропущенных значений
    missing = df.isnull().sum()
    if missing.any():
        issues.append(f"⚠️ Пропущенные значения: {missing[missing > 0].to_dict()}")
    
    # 3. Проверка отрицательных цен/количества
    if 'price' in df.columns and (df['price'] < 0).any():
        issues.append("❌ Найдены отрицательные цены!")
    
    if 'quantity' in df.columns and (df['quantity'] < 0).any():
        issues.append("❌ Найдено отрицательное количество!")
    
    # 4. Проверка логики: total = price * quantity
    if all(col in df.columns for col in ['price', 'quantity', 'total']):
        calculated_total = df['price'] * df['quantity']
        mismatch = abs(df['total'] - calculated_total) > 0.01
        if mismatch.any():
            issues.append(f"⚠️ Несоответствие в {mismatch.sum()} строках: total ≠ price × quantity")
    
    return {
        'is_valid': len(issues) == 0,
        'issues': issues,
        'quality_score': 100 - (len(issues) * 10)
    }
```

### 1.2 Автоматическое исправление данных
```python
def auto_fix_data(df):
    """Автоматическое исправление данных"""
    fixed = []
    
    # 1. Удаление дубликатов
    before = len(df)
    df = df.drop_duplicates()
    after = len(df)
    if before != after:
        fixed.append(f"✅ Удалено {before - after} дубликатов")
    
    # 2. Заполнение пропущенных значений
    if 'quantity' in df.columns:
        df['quantity'].fillna(1, inplace=True)
        fixed.append("✅ Заполнены пропущенные количества (1)")
    
    # 3. Пересчёт total
    if all(col in df.columns for col in ['price', 'quantity']):
        df['total'] = df['price'] * df['quantity']
        fixed.append("✅ Пересчитаны все суммы (total = price × quantity)")
    
    # 4. Удаление отрицательных значений
    if 'price' in df.columns:
        negative_prices = (df['price'] < 0).sum()
        if negative_prices > 0:
            df = df[df['price'] >= 0]
            fixed.append(f"✅ Удалено {negative_prices} строк с отрицательными ценами")
    
    return df, fixed
```

def double_check_calculations(analytics):
    """Двойная проверка всех расчётов"""
    checks = []
    
    # 1. Проверка выручки
    calculated_revenue = sum([p['revenue'] for p in analytics['top_products']])
    if abs(calculated_revenue - analytics['total_revenue']) > 0.01:
        checks.append({
            'metric': 'Выручка',
            'status': 'error',
            'message': f"Несоответствие: {calculated_revenue} vs {analytics['total_revenue']}"
        })
    else:
        checks.append({
            'metric': 'Выручка',
            'status': 'ok',
            'message': '✅ Расчёт верен'
        })
    
    # 2. Проверка среднего чека
    calculated_avg = analytics['total_revenue'] / analytics['total_orders']
    if abs(calculated_avg - analytics['average_check']) > 0.01:
        checks.append({
            'metric': 'Средний чек',
            'status': 'error',
            'message': f"Несоответствие: {calculated_avg} vs {analytics['average_check']}"
        })
    else:
        checks.append({
            'metric': 'Средний чек',
            'status': 'ok',
            'message': '✅ Расчёт верен'
        })
    
    return checks
```

import anthropic

async def verify_with_claude(data, analytics):
    """Claude AI проверяет точность расчётов"""
    
    client = anthropic.Anthropic(api_key=settings.ANTHROPIC_API_KEY)
    
    prompt = f"""
    Проверь точность расчётов:
    
    Данные:
    - Всего строк: {len(data)}
    - Общая выручка: {analytics['total_revenue']}
    - Заказов: {analytics['total_orders']}
    - Средний чек: {analytics['average_check']}
    
    Проверь:
    1. Средний чек = Выручка / Заказы
    2. Нет ли аномалий в данных
    3. Логичны ли цифры
    
    Верни JSON:
    {{
        "is_correct": true/false,
        "confidence": 0-100,
        "issues": ["список проблем"],
        "recommendations": ["рекомендации"]
    }}
    """
    
    response = client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1000,
        messages=[{"role": "user", "content": prompt}]
    )
    
    return response.content[0].text
```
