ЕСЛИ ПОЯВЛЯЕТСЯ ОШИБКА ПОЛИТИКИ:

.\запуск.ps1


РУЧНОЙ ЗАПУСК (БЕЗ СКРИПТА):
----------------------------

# 1. Активировать venv и запустить бэкенд
cd "C:\Users\jobir\OneDrive\Desktop\Дашборд SaaS"
.\venv\Scripts\activate.ps1
uvicorn app.main:app --reload --host 0.0.0.0 --port 8000

# 2. В ДРУГОМ окне PowerShell - запустить фронтенд
cd "C:\Users\jobir\OneDrive\Desktop\Дашборд SaaS\frontend"
npm run dev

# 3. Подождать 30-60 секунд, затем открыть браузеры
Start-Process msedge "http://localhost:3000"
Start-Process msedge "http://localhost:8000/docs"


ПРОВЕРКА СЕРВЕРОВ:
------------------
# Проверить бэкенд
Invoke-WebRequest -Uri "http://localhost:8000/docs" -UseBasicParsing

# Проверить фронтенд
Invoke-WebRequest -Uri "http://localhost:3000" -UseBasicParsing


ОТКРЫТЬ БРАУЗЕРЫ:
-----------------
Start-Process msedge "http://localhost:3000"
Start-Process msedge "http://localhost:8000/docs"


ОСТАНОВИТЬ СЕРВЕРЫ:
-------------------
# Найти процессы
Get-Process python,node

# Остановить все
Stop-Process -Name python,node -Force
